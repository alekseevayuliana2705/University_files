@startuml
scale 2/1

skinparam monochrome true
skinparam defaultFontName Arial
skinparam usecase {
  BackgroundColor #F5F5F5
  BorderColor #333
}

actor Бизнес_аналитик <<Пользователь>> as ba
actor Системный_аналитик <<Пользователь>> as sa

rectangle "Система анализа данных" {
  usecase "Оценка качества данных" as UC1
  usecase "Выявление узких мест" as UC2
  usecase "Формирование гипотез оптимизации" as UC3
  usecase "Сравнительный анализ версий системы" as UC4
  usecase "Прогнозная оценка качества" as UC5

  UC1 --> UC2 : "На основе"
  UC2 --> UC3 : "Позволяет"
  UC4 --> UC1 : "Использует методы"
  UC3 --> UC5 : "Влияет на"
}

database "Источник данных\n(PostgreSQL)" as db {
  folder "Данные технологических окон" as data
}

ba --> UC1 : Работает с\nвизуализациями
sa --> UC2 : Анализирует\nаномалии
sa --> UC3 : Разрабатывает\nтребования
ba --> UC4 : Сравнивает\nпериоды
sa --> UC5 : Оценивает\nсценарии

UC1 --> data : Запрашивает данные
UC2 --> data : Анализирует данные
UC4 --> data : Сравнивает срезы данных
UC5 --> data : Моделирует\nсценарии

note right of UC4
  **Цель**: Повысить оперативность анализа
  при переходе на новую версию системы
  за счет автоматизации проверок
end note

note left of UC5
  **Перспектива**: Прогноз качества данных
  при изменении условий сбора
  с использованием ML-моделей
end note

@enduml
